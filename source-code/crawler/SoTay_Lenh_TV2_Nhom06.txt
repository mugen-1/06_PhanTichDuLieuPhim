# (1 lần) File mật khẩu + biến kết nối A lưu trong .bashrc
read -s -p "Mat khau MySQL cua tv2: " P; echo
echo -n "$P" > ~/.tv2.pwd; chmod 400 ~/.tv2.pwd; unset P
echo 'export A="--connect jdbc:mysql://localhost:3306/moviedb?useUnicode=true&characterEncoding=UTF-8&useSSL=false&allowPublicKeyRetrieval=true --username tv2 --password-file file://$HOME/.tv2.pwd"' >> ~/.bashrc
grep -q "file.encoding=UTF-8" ~/.bashrc || echo 'export JAVA_TOOL_OPTIONS="-Dfile.encoding=UTF-8"' >> ~/.bashrc
source ~/.bashrc

# Kiểm tra kết nối (dòng Warning HCatalog/Accumulo/Zookeeper là bình thường, đã lọc)
sqoop list-tables $A 2>/dev/null | grep -v -E "^Warning|^Please set"
sqoop eval $A --query "SELECT COUNT(*) FROM movies" 2>/dev/null | grep -v -E "^Warning|^Please set"

# Import 6 bảng (movie_people chạy 2 mapper)
hdfs dfs -mkdir -p /user/nhom06/movies/sqoop
for t in movies genres movie_genres people reviews; do
  sqoop import $A --table $t --target-dir /user/nhom06/movies/sqoop/$t --delete-target-dir \
    --fields-terminated-by '\t' --null-string '\\N' --null-non-string '\\N' -m 1 2>&1 | grep -E "Retrieved|ERROR"
done
sqoop import $A --table movie_people --target-dir /user/nhom06/movies/sqoop/movie_people --delete-target-dir \
  --fields-terminated-by '\t' --null-string '\\N' --null-non-string '\\N' --split-by movie_id -m 2 2>&1 | grep -E "Retrieved|ERROR"

# Kiểm tra
hdfs dfs -ls -R /user/nhom06/movies/sqoop | grep part-m
for t in movies movie_people; do echo -n "$t: "; hdfs dfs -cat /user/nhom06/movies/sqoop/$t/part-m-* | wc -l; done

 I. MAPREDUCE (PYTHON STREAMING) – chạy cùng 1 Terminal
======================================================================
cd ~/06_PhanTichDuLieuPhim
M=source-code/mapreduce
STREAM=$HADOOP_HOME/share/hadoop/tools/lib/hadoop-streaming-3.4.2.jar

# MR1 – lọc lỗi dữ liệu OMDb (chỉ có Map)
hdfs dfs -rm -r -f /user/nhom06/movies/mr/mr1_omdb_sach
hadoop jar $STREAM -D mapreduce.job.name="TV2 MR1 loc loi OMDb" -D mapreduce.job.reduces=0 \
  -files $M/tv2_mr1_loc_loi_omdb/mapper.py -mapper "python3 mapper.py" \
  -input "/user/nhom06/movies/raw/flume/*/omdb_movies.jsonl*" -output /user/nhom06/movies/mr/mr1_omdb_sach
hdfs dfs -cat /user/nhom06/movies/mr/mr1_omdb_sach/part-* | head -5

# MR2 – loại phim trùng theo imdb_id
hdfs dfs -rm -r -f /user/nhom06/movies/mr/mr2_tmdb_khong_trung
hadoop jar $STREAM -D mapreduce.job.name="TV2 MR2 loai trung imdb_id" \
  -files $M/tv2_mr2_loai_trung/mapper.py,$M/tv2_mr2_loai_trung/reducer.py \
  -mapper "python3 mapper.py" -reducer "python3 reducer.py" \
  -input /user/nhom06/movies/raw/tmdb/tmdb_movies.jsonl -output /user/nhom06/movies/mr/mr2_tmdb_khong_trung
hdfs dfs -cat /user/nhom06/movies/mr/mr2_tmdb_khong_trung/part-* | wc -l

# MR3 – chuẩn hóa thể loại
hdfs dfs -rm -r -f /user/nhom06/movies/mr/mr3_the_loai
hadoop jar $STREAM -D mapreduce.job.name="TV2 MR3 chuan hoa the loai" \
  -files $M/tv2_mr3_the_loai/mapper.py,$M/tv2_mr3_the_loai/reducer.py \
  -mapper "python3 mapper.py" -reducer "python3 reducer.py" \
  -input "/user/nhom06/movies/raw/flume/*/omdb_movies.jsonl*" -output /user/nhom06/movies/mr/mr3_the_loai
hdfs dfs -cat /user/nhom06/movies/mr/mr3_the_loai/part-* | cut -f2 | sort | uniq -c | sort -rn

OUT=dataset/tong_quan_du_lieu_tv2.txt

{
echo "===== TONG QUAN DU LIEU PHAN THANH VIEN 2 - cap nhat $(date '+%F %H:%M') ====="
echo; echo "########## 1. MYSQL - CSDL moviedb ##########"
$M -e "SELECT 'movies' bang, COUNT(*) so_dong FROM movies UNION ALL SELECT 'genres', COUNT(*) FROM genres
UNION ALL SELECT 'movie_genres', COUNT(*) FROM movie_genres UNION ALL SELECT 'people', COUNT(*) FROM people
UNION ALL SELECT 'movie_people', COUNT(*) FROM movie_people UNION ALL SELECT 'reviews', COUNT(*) FROM reviews;"
for t in movies genres movie_genres people movie_people reviews; do
  echo; echo "=== Bang $t: cau truc ==="
  $M -e "DESCRIBE $t;"
  echo "=== Bang $t: dong mau ==="
  if [ $t = movies ]; then mysql -u tv2 -p$(cat ~/.tv2.pwd) moviedb -e "SELECT * FROM movies ORDER BY imdb_votes DESC LIMIT 2\G"
  else $M -e "SELECT * FROM $t LIMIT 3;"; fi
done
echo; echo "########## 2. MONGODB - CSDL moviedb ##########"
mongosh moviedb --quiet --eval '
db.getCollectionNames().forEach(c => { print("\n=== Collection " + c + ": " + db[c].countDocuments() + " document. 1 document mau:"); printjson(db[c].findOne({}, {_id:0})); });'
echo; echo "########## 3. HDFS - /user/nhom06/movies ##########"
hdfs dfs -du -h /user/nhom06/movies/* 2>/dev/null

======================================================================
 N. LỖI ĐÃ GẶP VÀ CÁCH SỬA (ghi vào mục "Khó khăn" của báo cáo)
======================================================================
 1. Không vào được themoviedb.org      -> nhà mạng chặn DNS -> đổi DNS 1.1.1.1 / 8.8.8.8
 2. TMDB báo 401                        -> dán nhầm key OMDb (8 ký tự) -> dùng đúng API Key TMDB 32 ký tự
 3. OMDb dừng ở 1.000 phim              -> key miễn phí giới hạn lượt/ngày -> hôm sau chạy tiếp
 4. Wikidata lỗi 502                    -> máy chủ quá tải -> thử lại có chờ tăng dần, ghi theo nhóm 50 phim
 5. Scrapy ra 0 phim                    -> link Wikipedia dạng đầy đủ https://..., tên phim có dấu ":" -> sửa bộ chọn
 6. mongoimport lỗi "Character: N"      -> pandas ghi NaN, JSON không có NaN -> đổi NaN thành null
 7. MySQL "Loading local data is disabled" -> SET GLOBAL mất khi khởi động lại -> SET PERSIST local_infile = 1
 8. Flume OutOfMemoryError              -> heap mặc định 20 MB + nạp thư viện Hive -> flume-env.sh -Xmx512m, HIVE_HOME=
 9. mongosh đứng ở dấu ">"              -> thiếu dấu ' đóng lệnh -> Ctrl + C, dán lại đủ
10. YARN 8088 không mở được             -> nhóm cấu hình cổng 9004 trong yarn-site.xml
11. scp không chép đè được file mới    -> ghi file trực tiếp trong máy ảo bằng cat > file <<'EOF'
